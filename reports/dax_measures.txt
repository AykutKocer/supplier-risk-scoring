Power BI DAX measures for the Supplier Risk & Performance Scoring dashboard.
Paste each block into Power BI Desktop as a new measure (Home > New Measure)
on the supplier_risk_scores table.

--------------------------------------------------------------------------
Total Suppliers = COUNTROWS(supplier_risk_scores)

--------------------------------------------------------------------------
Average Risk Score = ROUND(AVERAGE(supplier_risk_scores[risk_score]), 1)

--------------------------------------------------------------------------
High Risk Count =
CALCULATE(
    COUNTROWS(supplier_risk_scores),
    supplier_risk_scores[risk_level] = "High"
)

--------------------------------------------------------------------------
High Risk % =
DIVIDE([High Risk Count], [Total Suppliers])

--------------------------------------------------------------------------
Suppliers With Estimated Data =
CALCULATE(
    COUNTROWS(supplier_risk_scores),
    supplier_risk_scores[has_estimated_inputs] = TRUE
)

--------------------------------------------------------------------------
Average Delivery Delay Rate = ROUND(AVERAGE(supplier_risk_scores[delivery_delay_rate]), 3)

--------------------------------------------------------------------------
Average Dependency Ratio = ROUND(AVERAGE(supplier_risk_scores[supplier_dependency_ratio]), 3)

--------------------------------------------------------------------------
-- v2 measures: financial risk axis (deliberately separate from risk_score,
-- see README "Scoring methodology" > Financial risk axis).

Average Financial Risk Score = ROUND(AVERAGE(supplier_risk_scores[financial_risk_score]), 1)

--------------------------------------------------------------------------
High Financial Risk Count =
CALCULATE(
    COUNTROWS(supplier_risk_scores),
    supplier_risk_scores[financial_risk_level] = "High"
)

--------------------------------------------------------------------------
High Financial Risk % =
DIVIDE([High Financial Risk Count], [Total Suppliers])

--------------------------------------------------------------------------
-- Suppliers that are High on BOTH axes at once — the ones needing the most
-- urgent attention, since operational and financial risk are independent.
High Risk On Both Axes =
CALCULATE(
    COUNTROWS(supplier_risk_scores),
    supplier_risk_scores[risk_level] = "High",
    supplier_risk_scores[financial_risk_level] = "High"
)

--------------------------------------------------------------------------
-- v2 measure: portfolio concentration (HHI). Sum of each supplier's squared
-- share of total purchase volume, scaled to the conventional 0-10,000 range
-- (see compute_portfolio_hhi() in src/score_suppliers.py; same DOJ
-- antitrust-review thresholds: <1,500 low, 1,500-2,500 moderate, >2,500 high).
Portfolio HHI =
SUMX(
    supplier_risk_scores,
    supplier_risk_scores[supplier_dependency_ratio] ^ 2
) * 10000

--------------------------------------------------------------------------
-- v2 measure: CBAM compliance exposure. Boolean applicability + gap flag,
-- not a score -- see compute_cbam_compliance_risk() in
-- src/score_suppliers.py and config/cbam_sectors.yaml.
CBAM Exposed Count =
CALCULATE(
    COUNTROWS(supplier_risk_scores),
    supplier_risk_scores[cbam_compliance_risk] = TRUE
)
